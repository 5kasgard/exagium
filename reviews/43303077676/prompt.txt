466: def _orphan_entries_problem(o, pred):
467:     """The orphan must publish the hidden entries: canonical, consecutive from from_seq, chained from prev_sha256,
468:     ending at head_sha256. Returns a description of the first problem, or None."""
469:     try:

230: def one_rep(cmd, cache, env, exe, inputs, timeout, keep_raw=None, single_process=False, confine=None):
231:     """Run cmd once under cachegrind in a fresh sandbox, children included. Counts summed over processes."""
232:     fresh_sandbox(inputs)
233:     with tempfile.TemporaryDirectory() as td:
234:         # One shared log stream on a pipe (valgrind keeps it on a reserved fd the program can't close). Each
235:         # process image valgrind starts writes one "Command:" header, so a pid with two headers exec'd. A pipe,
236:         # unlike a file, can't be edited afterwards: a header once written can't be removed (review 2).
237:         rfd, wfd = os.pipe()
238:         chunks = []
239:         reader = threading.Thread(target=lambda: chunks.extend(iter(lambda: os.read(rfd, 65536), b"")), daemon=True)
240:         reader.start()
241:         vg = [*confine_prefix(confine, td), exe, "--command-line-only=yes", "--tool=cachegrind", "--cache-sim=yes", "--trace-children=yes",
242:               f"--cachegrind-out-file={td}/cg.%p", f"--log-fd={wfd}", *cache_args(cache), "--", *cmd]
243:         pid0 = last_pid() if single_process else None
244:         t0 = datetime.datetime.now(datetime.timezone.utc)
245:         try:
246:             p = subprocess.run(vg, env=env, cwd=os.path.join(SANDBOX, "work"), capture_output=True,
247:                                timeout=timeout, pass_fds=(wfd,))
248:         finally:
249:             os.close(wfd)
250:         wall = (datetime.datetime.now(datetime.timezone.utc) - t0).total_seconds()
251:         if single_process and (made := last_pid() - pid0) != 1:
252:             # every process or thread takes a pid, even one killed before it reports its counts (review 3)
253:             raise Failure(f"the command made {made - 1} more process(es) or thread(s); --single-process allows none")
254:         reader.join(10)
255:         if reader.is_alive():  # someone still holds the log open: a process outlived the command
256:             raise Failure("a process outlived the command (its work would be counted late or not at all)")
257:         os.close(rfd)
258:         log = b"".join(chunks).decode(errors="replace")
259:         if p.returncode != 0:
260:             raise Failure(f"exit {p.returncode}: {log.strip()[-400:]}")
261:         # Cross-check the pipe (append-only) against the count files (which a program could edit or delete):
262:         # each process image writes one "Command:" header when valgrind starts it (a fork child doesn't), and
263:         # every process, fork children included, writes "I refs: N" when it ends.
264:         starts, irefs = {}, {}
265:         for line in log.splitlines():
266:             if not line.startswith("==") or line.find("==", 2) < 0:
267:                 continue
268:             pid, text = line[2:line.index("==", 2)], line[line.index("==", 2) + 2:].strip()
269:             if text.startswith("Command: "):
270:                 starts[pid] = starts.get(pid, 0) + 1
271:             elif text.startswith("I refs:"):
272:                 if pid in irefs:
273:                     raise Failure(f"process {pid} reported its total twice")
274:                 irefs[pid] = int(text.split(":", 1)[1].replace(",", ""))
275:         # Only the command itself may start a process image. A second header means an exec (by the command, or
276:         # by a fork child: valgrind drops the counts of any image before its exec, so work there would vanish).
277:         if sum(starts.values()) != 1:
278:             raise Failure(f"{sum(starts.values()) - 1} exec call(s) (processes {', '.join(sorted(starts))}): valgrind "
279:                           f"drops the counts of an image before it execs, so this could undercount. Only fork "
280:                           f"children are allowed; run programs directly, without exec, shells or subprocess spawns.")
281:         if set(starts) - set(irefs):
282:             raise Failure(f"process {', '.join(sorted(set(starts) - set(irefs)))} ended without reporting counts")
283:         outs = sorted(f for f in os.listdir(td) if f.startswith("cg."))
284:         if not outs or {f[3:] for f in outs} != set(irefs):
285:             raise Failure(f"{len(irefs)} processes reported totals but {len(outs)} left count files")
286:         if keep_raw:
287:             os.makedirs(keep_raw, exist_ok=True)
288:             for f in outs:
289:                 dst = os.path.join(keep_raw, f"{f}.{os.getpid()}.{len(os.listdir(keep_raw))}.gz")
290:                 with open(os.path.join(td, f), "rb") as s, gzip.open(dst, "xb") as d:
291:                     shutil.copyfileobj(s, d)
292:         total, models = dict.fromkeys(EVENTS, 0), []
293:         for f in outs:
294:             desc, counts = parse_out(os.path.join(td, f))
295:             if counts["Ir"] != irefs[f[3:]]:
296:                 raise Failure(f"count file of process {f[3:]} disagrees with the total it reported")
297:             models.append(desc)
298:             for k in EVENTS:
299:                 total[k] += counts[k]
300:         if any(m != models[0] for m in models):
301:             raise Failure("processes simulated different cache models")
302:         return {"counts": total, "processes": len(outs), "wall_s": round(wall, 3),
303:                 "stdout_sha256": hashlib.sha256(p.stdout).hexdigest(), "model": models[0]}

328: def count(cmd, reps=3, cache=REFERENCE, extra_env=(), input_paths=(), timeout=3600, keep_raw=None, single_process=False,
329:           confine=None):
330:     exe = shutil.which("valgrind")

332:         raise Failure("valgrind not found")
333:     if not cmd:
334:         raise Failure("no command")

336:     for kv in extra_env:
337:         k, sep, v = kv.partition("=")
338:         if not sep or not k.replace("_", "").isalnum() or k in ("PATH", "HOME", "TMPDIR") or k.startswith(DENY_ENV_PREFIXES):
339:             raise Failure(f"--env {kv}: must be KEY=VALUE, and not PATH, HOME, TMPDIR or {', '.join(DENY_ENV_PREFIXES)}*")
340:         env[k] = v

343:                       "them unfingerprinted (review 4)")

345:     fp, prog = fingerprint(cmd, inputs)

357:     if any(r["model"] != model for r in rows) or (requested and model != requested):
358:         raise Failure(f"valgrind simulated {model}, not the requested {requested}")
359:     crc = client_request_code(scanned)
360:     rec = {
361:         "schema": SCHEMA,
362:         "created_utc": datetime.datetime.now(datetime.timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ"),
363:         "command": list(cmd),
364:         "env": env,
365:         "inputs": {b: {"path": v["path"], "sha256": v["sha256"]} for b, v in inputs.items()},
366:         "executes": fp,
367:         "tool": {"valgrind": valgrind_version(exe), "files": tool_fingerprint(exe), "options": ["--tool=cachegrind", "--cache-sim=yes",
368:                  "--trace-children=yes", "--command-line-only=yes"] + (["exagium:--single-process"] if single_process else [])
369:                  + (["exagium:--confine"] if confine else []),
370:                  "sandbox": SANDBOX, **({"confine": {"sha256": sha256_file(confine["exe"]), "ro": confine["ro"], "ls": confine.get("ls", [])}} if confine else {})},
371:         "host": host(),
372:         "cache": {"requested": requested or "host", "simulated": model, "label": "chosen" if requested else "reported"},
373:         "reps": [{k: r[k] for k in ("counts", "processes", "wall_s", "stdout_sha256")} for r in rows],
374:         "summary": summarise(rows),
375:         "verdict": {"exact": all(r["counts"] == rows[0]["counts"] for r in rows),
376:                     "stdout_identical": len({r["stdout_sha256"] for r in rows}) == 1,
377:                     "client_request_code": crc, "reps": reps},
378:         "labels": {"counts": "simulated", "wall_s": "measured (under valgrind; not the program's native time)"},
379:         "caveats": [
380:             "Counts are for the CPU valgrind presents, not the host's: on x86 it offers at most AVX2, so code "
381:             "that dispatches on CPU features runs its AVX2-or-lower path here even where the host has AVX-512.",
382:             "The cache model is two levels (L1 and LL), LRU, no prefetching; miss counts are relative to that model, "
383:             "not predictions of this machine's latency.",
384:             "The sandbox is a fresh directory, HOME and TMPDIR, not a container: a program can still reach absolute "
385:             "paths elsewhere, other processes and the network. Run untrusted candidates inside a fresh container.",
386:             "client_request_code scans the program, interpreter, their linked libraries and inputs; code loaded "
387:             "later (dlopen, JIT) is not scanned.",

405: def _validate(rec):
406:     body = {k: v for k, v in rec.items() if k != "digest_sha256"}
407:     if hashlib.sha256(canonical(body)).hexdigest() != rec.get("digest_sha256"):
408:         return "record digest does not match its contents (edited after it was written)"
409:     if rec.get("schema") != SCHEMA:
410:         return OTHER_SCHEMA + f" ({rec.get('schema')!r}, this version reads {SCHEMA})"
411:     cmd, env, inputs, cache = rec["command"], rec["env"], rec["inputs"], rec["cache"]["requested"]
412:     if not (isinstance(cmd, list) and cmd and all(isinstance(c, str) for c in cmd)):
413:         return "command must be a non-empty list of strings"
414:     if not (isinstance(env, dict) and all(isinstance(k, str) and isinstance(v, str) for k, v in env.items())):
415:         return "env must map strings to strings"
416:     if not (isinstance(inputs, dict) and all(isinstance(v, dict) and isinstance(v.get("sha256"), str) for v in inputs.values())):
417:         return "inputs must map names to {path, sha256}"
418:     if cache != "host" and not (isinstance(cache, dict) and set(cache) == {"I1", "D1", "LL"} and all(
419:             isinstance(v, list) and len(v) == 3 and all(isinstance(x, int) and x > 0 for x in v) for v in cache.values())):
420:         return "cache.requested must be 'host' or I1/D1/LL as [size, ways, line]"
421:     reps = rec["reps"]
422:     if not isinstance(reps, list) or not reps or len(reps) != rec["verdict"]["reps"]:
423:         return "rep list missing or wrong length"
424:     for i, r in enumerate(reps):
425:         if set(r["counts"]) != set(EVENTS) or not all(isinstance(x, int) and x >= 0 for x in r["counts"].values()):
426:             return f"rep {i}: wrong events or non-integer counts"
427:     if rec["verdict"]["exact"] != all(r["counts"] == reps[0]["counts"] for r in reps):
428:         return "verdict.exact disagrees with the recorded reps"
429:     if rec["verdict"]["stdout_identical"] != (len({r["stdout_sha256"] for r in reps}) == 1):
430:         return "verdict.stdout_identical disagrees with the recorded reps"
431:     if summarise(reps) != rec["summary"]:
432:         return "summary disagrees with the recorded reps"
433:     return None

436: def check(path, input_dir=None, reps=None):
437:     """Re-count a recorded command here and compare. Returns (exit code, message)."""
438:     try:
439:         with open(path) as f:
440:             rec = json.load(f)
441:         problem = _validate(rec)
442:     except (OSError, ValueError, KeyError, TypeError, AttributeError) as e:
443:         return 1, f"malformed record: {type(e).__name__}: {e}"
444:     if problem:
445:         # an intact record from another version is "can't compare" (2), not "edited" (1): Z, 2026-11-12
446:         return (2 if problem.startswith(OTHER_SCHEMA) else 1), problem
447:     try:
448:         return _recheck(rec, input_dir, reps)
449:     except (KeyError, TypeError, AttributeError, ValueError) as e:
450:         return 1, f"malformed record: {type(e).__name__}: {e}"

453: def _recheck(rec, input_dir, reps):
454:     paths = []
455:     for base, v in rec["inputs"].items():
456:         p = os.path.join(input_dir, base) if input_dir else v["path"]
457:         if not os.path.isfile(p) or sha256_file(p) != v["sha256"]:
458:             return 2, f"input {base} is missing or differs at {p}"
459:         paths.append(p)
460:     exe = shutil.which("valgrind")
461:     if not exe or valgrind_version(exe) != rec["tool"]["valgrind"]:
462:         return 2, "valgrind missing or a different version"
463:     tf = tool_fingerprint(exe)
464:     if {k: tf[k] for k in ("launcher", "cachegrind")} != {k: rec["tool"]["files"][k] for k in ("launcher", "cachegrind")}:
465:         return 2, "a different valgrind launcher or cachegrind binary"
466:     if not rec["verdict"]["exact"] or rec["verdict"]["client_request_code"] != []:
467:         return 2, "the recorded run was not a clean exact count (reps differ or counting could be switched off)"
468:     h = host()
469:     if h["machine"] != rec["host"]["machine"]:
470:         return 2, "different architecture"
471:     if h["ld_so_cache_sha256"] != rec["host"]["ld_so_cache_sha256"]:
472:         return 2, "a different loader cache (another system image): counts may legitimately differ"
473:     try:
474:         fp, _ = fingerprint(rec["command"], resolve_inputs(paths))
475:     except Failure as e:
476:         return 2, str(e)
477:     if fp != rec["executes"]:
478:         return 2, "the program, interpreter or a shared library differs from the recorded one"
479:     extra = [f"{k}={v}" for k, v in rec["env"].items() if k not in ("PATH", "HOME", "TMPDIR")]
480:     new = count(rec["command"], reps or len(rec["reps"]), rec["cache"]["requested"], extra, paths)
481:     if new["cache"]["simulated"] != rec["cache"]["simulated"]:
482:         return 2, "a different cache model was simulated (host cache differs)"
483:     diffs = []
484:     old_r, new_r = rec["reps"][0], new["reps"][0]
485:     for k in EVENTS:
486:         if old_r["counts"][k] != new_r["counts"][k]:
487:             diffs.append(f"{k} {old_r['counts'][k]}->{new_r['counts'][k]}")
488:     for k in ("processes", "stdout_sha256"):
489:         if old_r[k] != new_r[k]:
490:             diffs.append(f"{k} differs")
491:     if new["verdict"]["client_request_code"] != []:
492:         diffs.append("code that can switch counting off is now present")
493:     if not diffs and new["verdict"]["exact"]:
494:         return 0, "identical counts and output"
495:     return 1, (f"differs: {', '.join(diffs) or 'reps not exact'} "
496:                f"(recorded exact={rec['verdict']['exact']}, now exact={new['verdict']['exact']})")
