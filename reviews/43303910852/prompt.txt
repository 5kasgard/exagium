559: def cost_disagreements(pairs):
560:     """Pairs where the two cost meters point opposite ways: a uses fewer simulated instructions than b, yet a's native
561:     time per image is longer by more than 2 SE (paired, interleaved rounds). Simulated instructions miss dependency
562:     chains, mispredicted branches and cache behaviour, so a cost claim that rests on them alone can be wrong
563:     (system review, 2026-11-11: mlp64 used 0.43x linear's instructions and was slower on all 7 hosts). Each
564:     disagreeing pair is listed once, from the side the instruction count favours."""
565:     out = []
566:     for p in pairs:
567:         se = p.get("ns_per_image_diff_se")
568:         if p["Ir_ratio"] < 1 and se is not None and p["ns_per_image_diff"] > 2 * se:
569:             out.append({"fewer_instructions": p["a"], "but_slower_than": p["b"], "Ir_ratio": p["Ir_ratio"],
570:                         "ns_per_image_diff": p["ns_per_image_diff"], "ns_per_image_diff_se": se})
571:     return out
