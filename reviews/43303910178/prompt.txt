46: # ---------------------------------------------------------------- level detection
47: # How memory levels are read off the latency curve. All of it is recorded in the profile.
48: #  - Physics first: latency cannot fall as the working set grows, so a point significantly slower
49: #    than the next kept size is interference and is excluded (and listed). If more than
50: #    MAX_EXCLUDED of the curve breaks this, no levels are fitted at all.
51: #  - Piecewise-linear fit of log latency on log size, BIC, each point weighted by its own noise
52: #    (bootstrap SE of the median), widened to Student-t tails because it comes from few reps.
53: #    Real plateaus drift a little (associativity, TLB), so a plateau is one sloped segment.
54: #  - A level changes little compared with the jump at each end: its fitted rise across its span
55: #    must be significantly less than LEVEL_RISE_RATIO of each edge jump (owner's 2-SE rule,
56: #    Bonferroni-corrected because breakpoints are chosen from the data), and so must the plain
57: #    spread of its medians, which the weighting cannot hide; and per unit of size its slope must

77: def t_cdf(t, df):
78:     """Student-t CDF by Simpson integration of the density (stdlib only)."""
79:     if t < 0:
80:         return 1 - t_cdf(-t, df)
81:     c = math.exp(math.lgamma((df + 1) / 2) - math.lgamma(df / 2)) / math.sqrt(df * math.pi)
82:     f = lambda x: c * (1 + x * x / df) ** (-(df + 1) / 2)
83:     n = 2000; h = t / n
84:     s = f(0) + f(t) + sum((4 if k % 2 else 2) * f(k * h) for k in range(1, n))
85:     return 0.5 + s * h / 3

114:     """Piecewise-linear fit of y on x (log latency on log size), weights w, segment count by

118:     memo = {}

120:     def chi2(i, j):
121:         """Weighted residual of the best line through points i..j, computed directly (centred,
122:         two-pass) so weights that differ by many orders of magnitude cannot cancel to zero."""
123:         if j - i < 2:
124:             return 0.0  # one or two points: a line fits exactly
125:         if (i, j) not in memo:
126:             ws, xs, ys = w[i:j + 1], x[i:j + 1], y[i:j + 1]
127:             W = sum(ws); xm = sum(a * b for a, b in zip(ws, xs)) / W; ym = sum(a * b for a, b in zip(ws, ys)) / W
128:             sxx = sum(a * (b - xm) ** 2 for a, b in zip(ws, xs))
129:             sl = sum(a * (b - xm) * (c - ym) for a, b, c in zip(ws, xs, ys)) / sxx if sxx > 0 else 0.0

441: def nvml_lib_power():
442:     """Read GPU power through libnvidia-ml directly; None if the library is absent."""
443:     import ctypes
444:     try:
445:         lib = ctypes.CDLL("libnvidia-ml.so.1")

544: def chase_sizes(max_bytes, min_bytes=4 << 10):
545:     """Geometric sweep, factor sqrt(2), rounded to 4 KiB. No cache sizes assumed."""
546:     out, s = [], float(min_bytes)
547:     while s <= max_bytes * 1.0001:
548:         v = int(round(s / 4096)) * 4096
549:         if not out or v != out[-1]:
550:             out.append(v)
551:         s *= math.sqrt(2)
552:     return out

555: def plan(env, caches, quick, min_ms, max_bytes=None):
556:     ncpu = env["logical_cpus"] or 1
557:     llc = max((c["size_bytes"] for c in caches), default=32 << 20)
558:     avail = env["mem_available_bytes"] or (2 << 30)
559:     # Reported LLC is only a hint for how far to sweep (to pass it into DRAM). It is a VM's
560:     # claim and is often the whole host's cache, so we sweep 4x past it, capped by RAM.
561:     cap = max_bytes or min(max(4 * llc, 256 << 20), avail // 4)
562:     if quick:
563:         cap = min(cap, 256 << 20)
564:     sizes = chase_sizes(cap)
565:     if quick:
566:         sizes = sizes[::2] + ([sizes[-1]] if len(sizes) % 2 == 0 else [])
567:     line = next((c["line_bytes"] for c in caches if c["line_bytes"]), 64)
568:     m = []
569:     for s in sizes:
570:         m.append((f"chase:{s}", ["chase", str(s), str(line), str(min_ms), "1"]))
571:     stream_big = max(cap, 64 << 20)
572:     for th in sorted({1, ncpu}):
573:         m.append((f"stream:dram:{th}", ["stream", str(stream_big), str(th), str(min_ms * 3), "1"]))
574:         m.append((f"stream:l1:{th}", ["stream", str((16 << 10) * th), str(th), str(min_ms), "1"]))
575:         for op in ("scalar_f64", "vec_f64", "vec_f32"):
576:             m.append((f"flops:{op}:{th}", ["flops", op, str(th), str(min_ms * 3)]))
577:     m.append(("timer", ["timer"]))
578:     return m, {"sweep_max_bytes": cap, "line_bytes": line, "llc_reported_bytes": llc}

585: # ---------------------------------------------------------------- consistency
586: def check_raw(raw, settings):
587:     """Internal consistency of raw records: rep counts match settings, and each rate equals
588:     its own work divided by its own time. This is NOT proof the raw data is genuine;
589:     that needs the anchored record (open item 3)."""
590:     bad = []
591:     for name, recs in raw.items():
592:         if len(recs) != settings["reps"]:
593:             bad.append(f"{name}: {len(recs)} reps, settings say {settings['reps']}")
594:         for i, x in enumerate(recs):
595:             k, t = x["kind"], x.get("seconds")
596:             if k == "timer":
597:                 continue
598:             if not isinstance(t, (int, float)) or t <= 0:
599:                 bad.append(f"{name} rep {i}: missing or non-positive seconds, rate cannot be checked")
600:                 continue
601:             if k == "chase":
602:                 want, got = t / x["steps"] * 1e9, x["ns_per_load"]
603:             elif k == "stream":
604:                 want, got = x["bytes"] * x["passes"] / t / 1e9, x["GBps"]
605:             else:
606:                 want, got = 2 * x["lanes"] * x["chains"] * x["iters"] * x["threads"] / t / 1e9, x["GFLOPs"]
607:             if abs(want - got) > 1e-3 * abs(want) + 1e-3:
608:                 bad.append(f"{name} rep {i}: rate {got} but work/time gives {want:.4f}")
609:             floor = settings["min_ms"] * (1 if k == "chase" or name.startswith("stream:l1") else 3) / 1e3
610:             if t < floor * 0.999:
611:                 bad.append(f"{name} rep {i}: timed window {t:.4f}s is shorter than the {floor:.4f}s minimum")
612:             if k == "chase" and x.get("seed") != chase_seed(settings["seed"], i, x["bytes"]):
613:                 bad.append(f"{name} rep {i}: chase seed does not match the recorded run seed")
614:     return bad

629: def add_digests(profile):
630:     """measurement_sha256 covers what was measured and how (raw reps, settings, build, environment,
631:     meter tiers, time); profile_sha256 covers everything else as well. Either is what an outside
632:     record anchors: once anchored, any later edit to the profile shows as a different digest.
633:     A digest proves the profile was not changed after it was anchored, never that it is genuine."""
634:     body = {k: v for k, v in profile.items() if k != "digests"}
635:     meas = {k: profile[k] for k in ("raw", "settings", "build", "environment", "meter_tiers", "created_utc", "elapsed_s")}
636:     profile["digests"] = {"measurement_sha256": hashlib.sha256(canonical(meas)).hexdigest(),
637:                           "profile_sha256": hashlib.sha256(canonical(body)).hexdigest(),
638:                           "scheme": DIGEST_SCHEME, "label": "derived"}
639:     return profile

643: def analyse(raw, env, caches, tier_list, build_info, settings, elapsed):
644:     """Everything in the profile except raw/env/tiers is computed here from raw reps,
645:     so `--reanalyse` can regenerate (and check) any stored profile."""
646:     def stat(name, key):
647:         return summarise([x[key] for x in raw[name]])
648: 
649:     sizes = sorted(int(n.split(":")[1]) for n in raw if n.startswith("chase:"))
650:     lat = [stat(f"chase:{s}", "ns_per_load") for s in sizes]
651:     meds = [x["median"] for x in lat]
652:     df = min(x["n"] for x in lat) - 1 if lat else None
653:     ses = [x["se_median"] for x in lat]
654:     noise_known = bool(lat) and all(s is not None and s > 0 for s in ses)
655:     excluded = [sizes[k] for k in nonmonotone_points(meds, ses, df)] if noise_known else []
656:     levels = describe_levels(sizes, meds, ses, df=df,
657:                              rep_max=[x["max"] for x in lat], rep_min=[x["min"] for x in lat])
658:     if not noise_known:
659:         level_note = "not computed: needs >= 2 reps per size for noise estimates"
660:     elif levels is None:
661:         level_note = (f"not computed: {len(excluded)} of {len(sizes)} sizes are significantly slower than a larger "
662:                       f"size (more than {MAX_EXCLUDED:.0%}), so the curve is too disturbed to read levels from")
663:     else:
664:         level_note = LEVEL_METHOD
665:     llc = settings.get("llc_reported_bytes") or 0
666:     reach = sizes[-1] / llc if llc and sizes else None
667: 
668:     ncpu = env["logical_cpus"] or 1
669:     thr = sorted({1, ncpu})
670:     bw = {f"{k}_{th}t": {**stat(f"stream:{k}:{th}", "GBps"), "unit": "GB/s", "label": "measured",
671:                          "bytes": raw[f"stream:{k}:{th}"][0]["bytes"]}
672:           for k in ("dram", "l1") for th in thr}
673:     fl = {f"{op}_{th}t": {**stat(f"flops:{op}:{th}", "GFLOPs"), "unit": "GFLOP/s", "label": "measured"}

677:              "meaning": "min_step_ns is the smallest gap two reads can show (bounded by read cost), "
678:                         "so timed windows should be >> it; getres_ns_reported is the kernel's claim"}

738:     ap.add_argument("--reps", type=int, default=None, help="repetitions per measurement (default 7, quick 3)")

741:     ap.add_argument("--quick", action="store_true")
742:     ap.add_argument("--seed", type=int, default=None, help="shuffle seed (recorded)")
743:     ap.add_argument("--calibration", default=None, help="tier-3 calibration file, if any")
744:     ap.add_argument("--out", default="profile.json")
745:     ap.add_argument("--reanalyse", metavar="PROFILE", help="recompute all statistics from a profile's raw reps; "
746:                     "exit 1 if stored numbers differ or raw is internally inconsistent; exit 2 if consistent but "
747:                     "run on another host, so meter claims are unverified (consistency, never authenticity)")
748:     a = ap.parse_args(argv)
749:     if a.reanalyse:
750:         return reanalyse(a.reanalyse, a.out if a.out != "profile.json" else None)
751:     reps = a.reps or (3 if a.quick else 7)
752:     min_ms = a.min_ms or (10.0 if a.quick else 30.0)
753:     seed = a.seed if a.seed is not None else int(time.time())
754: 
755:     t_start = time.time()
756:     env, caches = environment(), reported_caches()
757:     tier_list = tiers(calibration=a.calibration)
758:     with tempfile.TemporaryDirectory() as td:
759:         exe, build_info = build(td)
760:         meas, plan_info = plan(env, caches, a.quick, min_ms, a.max_bytes)
761:         rnd = random.Random(seed)
762:         raw = {name: [] for name, _ in meas}
763:         steal = []
764:         for r in range(reps):
765:             t0, s0 = cpu_ticks()
766:             order = meas[:]
767:             rnd.shuffle(order)  # interleave: drift spreads over every measurement, not one
768:             for name, args in order:
769:                 if args[0] == "chase":  # a fresh random permutation each rep, so its effect shows up as noise
770:                     args = [*args, str(chase_seed(seed, r, int(args[1])))]
771:                 rc, out, err = run([exe, *args], timeout=600)
772:                 if rc != 0:
773:                     sys.exit(f"{name} failed: {err}")
774:                 rec = json.loads(out)
775:                 raw[name].append(rec)
776:             t1, s1 = cpu_ticks()
777:             steal.append(round((s1 - s0) / (t1 - t0), 4) if t0 is not None and t1 > t0 else None)
778:             print(f"rep {r + 1}/{reps} done ({time.time() - t_start:.0f}s, steal {steal[-1]})", file=sys.stderr)
779: 
780:     profile = analyse(raw, env, caches, tier_list, build_info,
781:                       {"reps": reps, "min_ms": min_ms, "seed": seed, "quick": a.quick,
782:                        "order": "shuffled per rep, all measurements interleaved",
783:                        "cpu_steal_fraction_per_rep": steal, "calibration": a.calibration, **plan_info},
784:                       round(time.time() - t_start, 1))
785:     add_digests(profile)
786:     with open(a.out, "w") as f:
787:         json.dump(profile, f, indent=1, allow_nan=False)
788:     print(summary_text(profile))
789:     print(f"\nprofile digest (anchor this): {profile['digests']['profile_sha256']}")
790:     print(f"\nwrote {a.out}", file=sys.stderr)

2: // Prints one JSON object on stdout. The driver (probe.py) repeats and interleaves.

4: //   kernels timer

6: //   kernels stream <bytes> <threads> <min_ms> <hugepage 0|1>       -> GB/s read
7: //   kernels flops  <scalar_f64|vec_f64|vec_f32> <threads> <min_ms> -> GFLOP/s (FMA = 2 flops)

19: static double now(void) {
20:   struct timespec t;
21:   clock_gettime(CLOCK_MONOTONIC, &t);
22:   return (double)t.tv_sec + (double)t.tv_nsec * 1e-9;

25: static void *big_alloc(size_t bytes, int huge) {
26:   size_t align = 2u << 20;
27:   size_t sz = (bytes + align - 1) / align * align;
28:   void *p = aligned_alloc(align, sz);
29:   if (!p) { fprintf(stderr, "alloc failed\n"); exit(2); }
30: #ifdef MADV_HUGEPAGE
31:   if (huge) madvise(p, sz, MADV_HUGEPAGE);
32: #endif

39: // ---------- timer ----------
40: static int cmd_timer(void) {
41:   // resolution: smallest nonzero step between consecutive reads; overhead: mean cost of a read
42:   double best = 1e9, t0 = now();
43:   for (int i = 0; i < 100000; i++) {
44:     double a = now(), b = now();
45:     while (b == a) b = now();
46:     if (b - a < best) best = b - a;
47:   }
48:   int n = 1000000; double s = now();
49:   for (int i = 0; i < n; i++) (void)now();
50:   double ovh = (now() - s) / n;
51:   struct timespec res; clock_getres(CLOCK_MONOTONIC, &res);
52:   // min_step_ns: smallest observable gap between two reads (bounded below by read cost);
53:   // getres_ns: what the kernel reports as the clock's resolution.
54:   printf("{\"kind\":\"timer\",\"min_step_ns\":%.3f,\"read_overhead_ns\":%.3f,\"getres_ns\":%.3f,\"seconds\":%.6f}\n",
55:          best * 1e9, ovh * 1e9, res.tv_sec * 1e9 + res.tv_nsec, now() - t0);
56:   return 0;
57: }

59: // ---------- latency: dependent pointer chase over a random single cycle of cache lines ----------
60: static int cmd_chase(size_t bytes, size_t line, double min_ms, int huge, uint64_t seed) {
61:   if (seed) rng = seed;
62:   if (line < sizeof(void *)) line = sizeof(void *);
63:   size_t n = bytes / line;
64:   if (n < 2) n = 2;
65:   char *buf = big_alloc(n * line, huge);
66:   size_t *perm = malloc(n * sizeof(size_t));
67:   for (size_t i = 0; i < n; i++) perm[i] = i;
68:   // Sattolo's algorithm: uniform random single cycle, so every line is visited once per lap
69:   for (size_t i = n - 1; i > 0; i--) { size_t j = xs() % i; size_t t = perm[i]; perm[i] = perm[j]; perm[j] = t; }
70:   for (size_t i = 0; i < n; i++) *(void **)(buf + i * line) = buf + perm[i] * line;
71:   free(perm);
72:   void **p = (void **)buf;
73:   for (size_t k = 0; k < n; k++) p = (void **)*p;  // warm: one full lap
74:   size_t steps = 1 << 16; double dt = 0;
75:   for (;;) {
76:     double t0 = now();
77:     for (size_t k = 0; k < steps; k++) p = (void **)*p;
78:     dt = now() - t0;
79:     if (dt * 1e3 >= min_ms) break;
80:     steps *= 2;
81:   }
82:   printf("{\"kind\":\"chase\",\"bytes\":%zu,\"line\":%zu,\"seed\":%llu,\"steps\":%zu,\"seconds\":%.9f,\"ns_per_load\":%.4f,\"sink\":%d}\n",
83:          n * line, line, (unsigned long long)seed, steps, dt, dt / steps * 1e9, p == NULL);
84:   free(buf);
85:   return 0;
86: }

96: static void *stream_worker(void *vp) {
97:   stream_arg *a = vp;
98:   const v8d *v = (const v8d *)a->base; size_t nv = a->bytes / sizeof(v8d);
99:   // 8 independent accumulators: enough to cover vector-add latency, so L1-resident
100:   // reads are limited by loads, not by the add chain (4 was add-bound: ~1 vector/cycle).
101:   v8d s0 = {0}, s1 = {0}, s2 = {0}, s3 = {0}, s4 = {0}, s5 = {0}, s6 = {0}, s7 = {0};
102:   pthread_barrier_wait(a->bar);
103:   for (size_t r = 0; r < a->passes; r++)
104:     for (size_t i = 0; i + 8 <= nv; i += 8) {
105:       s0 += v[i]; s1 += v[i + 1]; s2 += v[i + 2]; s3 += v[i + 3];
106:       s4 += v[i + 4]; s5 += v[i + 5]; s6 += v[i + 6]; s7 += v[i + 7];
107:     }
108:   pthread_barrier_wait(a->bar);
109:   v8d s = s0 + s1 + s2 + s3 + s4 + s5 + s6 + s7; double t = 0;
110:   for (int k = 0; k < 8; k++) t += s[k];
111:   a->sink = t;
112:   return NULL;
113: }

115: static double stream_once(char *buf, size_t bytes, int th, size_t passes, double *sink) {
116:   pthread_t tid[256]; stream_arg arg[256]; pthread_barrier_t bar;
117:   pthread_barrier_init(&bar, NULL, th + 1);
118:   size_t chunk = bytes / th / 512 * 512;
119:   for (int t = 0; t < th; t++) {
120:     arg[t] = (stream_arg){buf + t * chunk, chunk, passes, &bar, 0};
121:     pthread_create(&tid[t], NULL, stream_worker, &arg[t]);
122:   }
123:   pthread_barrier_wait(&bar); double t0 = now();
124:   pthread_barrier_wait(&bar); double dt = now() - t0;
125:   for (int t = 0; t < th; t++) { pthread_join(tid[t], NULL); *sink += arg[t].sink; }
126:   pthread_barrier_destroy(&bar);
127:   return dt;
128: }

130: static int cmd_stream(size_t bytes, int th, double min_ms, int huge) {
131:   if (th < 1 || th > 256) th = 1;
132:   char *buf = big_alloc(bytes, huge);
133:   double *d = (double *)buf;
134:   for (size_t i = 0; i < bytes / sizeof(double); i++) d[i] = (double)(i & 1023) * 0.5;
135:   size_t chunk = bytes / th / 512 * 512, passes = 1; double sink = 0, dt;
136:   stream_once(buf, bytes, th, 1, &sink);  // warm
137:   for (;;) {

141:   }
142:   double moved = (double)chunk * th * passes;
143:   printf("{\"kind\":\"stream\",\"bytes\":%zu,\"threads\":%d,\"passes\":%zu,\"seconds\":%.9f,\"GBps\":%.4f,\"sink\":%d}\n",
144:          chunk * th, th, passes, dt, moved / dt / 1e9, sink == 1.2345);
145:   free(buf);
146:   return 0;
147: }

150: #define NACC 16
151: __attribute__((optimize("no-tree-vectorize", "no-tree-slp-vectorize")))
152: static double flops_scalar(size_t it) {
153:   double x[NACC]; for (int j = 0; j < NACC; j++) x[j] = 1.0 + j * 1e-3;
154:   const double m = 0.9999999, c = 1e-9;
155:   for (size_t i = 0; i < it; i++)
156: #pragma GCC unroll 16
157:     for (int j = 0; j < NACC; j++) x[j] = x[j] * m + c;
158:   double s = 0; for (int j = 0; j < NACC; j++) s += x[j];
159:   return s;
160: }
161: static double flops_vf64(size_t it) {
162:   v8d x[NACC]; for (int j = 0; j < NACC; j++) for (int k = 0; k < 8; k++) x[j][k] = 1.0 + j * 1e-3 + k * 1e-4;
163:   const v8d m = {0.9999999,0.9999999,0.9999999,0.9999999,0.9999999,0.9999999,0.9999999,0.9999999};
164:   const v8d c = {1e-9,1e-9,1e-9,1e-9,1e-9,1e-9,1e-9,1e-9};
165:   for (size_t i = 0; i < it; i++)
166: #pragma GCC unroll 16
167:     for (int j = 0; j < NACC; j++) x[j] = x[j] * m + c;
168:   double s = 0; for (int j = 0; j < NACC; j++) for (int k = 0; k < 8; k++) s += x[j][k];
169:   return s;
170: }
